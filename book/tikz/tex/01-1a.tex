\begin{tikzpicture}
\begin{axis}[
    axis lines = middle,
    xlabel = $t$,
    xlabel style={at={(axis cs:5.8,0)}, anchor=west},
    ylabel = $f(t)$,
    xmin = -1, xmax = 5.8,
    ymin = -0.15, ymax = 1.3,
    domain = -0.8:5.6,
    samples = 150,
    xtick = {0.9,3.2},
    xticklabels = {$a$,$x$},
    ytick = \empty,
    clip = false,
]

% Aire hachurée noire : intégrale de a à x
\addplot[
    domain = 0.9:3.2,
    hachure,
] {1/(1+x^2)} \closedcycle;

% Aire hachurée grise : intégrale de x à +infini (même sens de hachures, couleur différente)
\addplot[
    domain = 3.2:5.6,
    hachure = monGris,
] {1/(1+x^2)} \closedcycle;

% La courbe elle-même
\addplot[courbe ] {1/(1+x^2)};

% Repères verticaux
\draw[dashed] (axis cs:0.9,0) -- (axis cs:0.9,{1/(1+0.9^2)});
\draw[dashed] (axis cs:3.2,0) -- (axis cs:3.2,{1/(1+3.2^2)});

% Labels
\node at (axis cs:4.0,0.7) {$\displaystyle{\color{black}\int_a^x f(t)\,dt} \xrightarrow[x\to+\infty]{} {\color{gray!60}\int_a^{+\infty} f(t)\,dt}$};

\end{axis}
\end{tikzpicture}
